Practical 1: Count Vowels and Consonants

Aim: Write a LEX program to count the number of vowels and consonants in a given string.

%{ #include int vowel_count = 0; int consonant_count = 0; %} %% [aeiouAEIOU] { vowel_count++; } [a-zA-Z] { consonant_count++; } . { ; } \n { ; } %% int yywrap() { return 1; } int main() { printf("Enter a string: "); yylex(); printf("Number of vowels: %d\n", vowel_count); printf("Number of consonants: %d\n", consonant_count); return 0; } /* Compilation & Execution: lex filename.l gcc lex.yy.c ./a.out */

Practical 4: Count and Remove Comments from C Program

Aim: Count the number of comment lines in a C program, eliminate them, and copy the program to a separate file.

Steps:

  1. Save LEX program as prac4.l
  2. Create input C file: comment.c
  3. Output will be saved to output.c
%{ #include int comment_count = 0; FILE *output_file; %} %% "//".* { comment_count++; } "/*"([^*]|\*+[^*/])*\*+"/" { comment_count++; } . { fprintf(output_file, "%s", yytext); } \n { fprintf(output_file, "\n"); } %% int yywrap() { return 1; } int main(int argc, char *argv[]) { if(argc != 2) { printf("Usage: ./a.out \n"); return 1; } yyin = fopen(argv[1], "r"); if(!yyin) { printf("Error opening input file\n"); return 1; } output_file = fopen("output.c", "w"); if(!output_file) { printf("Error creating output file\n"); return 1; } yylex(); printf("Number of comment lines: %d\n", comment_count); printf("Program without comments saved to output.c\n"); fclose(yyin); fclose(output_file); return 0; } /* Compilation & Execution: lex prac4.l gcc lex.yy.c ./a.out comment.c */

Practical 5: Replace scanf/printf with readf/writef

Aim: Count the number of 'scanf' and 'printf' statements in a C program and replace them with 'readf' and 'writef' respectively.

%{ #include int scanf_count = 0; int printf_count = 0; FILE *output_file; %} %% "scanf" { scanf_count++; fprintf(output_file, "readf"); } "printf" { printf_count++; fprintf(output_file, "writef"); } . { fprintf(output_file, "%s", yytext); } \n { fprintf(output_file, "\n"); } %% int yywrap() { return 1; } int main(int argc, char *argv[]) { if(argc != 2) { printf("Usage: ./a.out \n"); return 1; } yyin = fopen(argv[1], "r"); if(!yyin) { printf("Error opening input file\n"); return 1; } output_file = fopen("modified.c", "w"); if(!output_file) { printf("Error creating output file\n"); return 1; } yylex(); printf("Number of scanf statements: %d\n", scanf_count); printf("Number of printf statements: %d\n", printf_count); printf("Modified program saved to modified.c\n"); fclose(yyin); fclose(output_file); return 0; } /* Compilation & Execution: lex prac5.l gcc lex.yy.c ./a.out arithmetic.c */

Practical 7: Recognize Simple or Compound Sentence

Practical 9: Validate Simple Expression

Aim: Test the validity of a simple expression involving operators +, -, *, /.

%{ #include int valid = 1; %} %% [a-zA-Z][a-zA-Z0-9]* { ; } [0-9]+ { ; } [\+\-\*\/] { ; } [\(\)] { ; } [ \t] { ; } \n { return 0; } . { valid = 0; } %% int yywrap() { return 1; } int main() { printf("Enter an arithmetic expression:\n"); yylex(); if(valid == 1) { printf("Valid expression\n"); } else { printf("Invalid expression\n"); } return 0; } /* Compilation & Execution: lex prac9.l gcc lex.yy.c ./a.out */

Practical 10: Recognize Nested IF Statements

Aim: Recognize nested IF control statements and display the level of nesting.

Files Required:

  • work.y (YACC file)
  • work.l (LEX file)

YACC File (work.y)

%{ #include int level = 0; int max_level = 0; %} %token IF ELSE NUM ID %% stmt: if_stmt { printf("Maximum nesting level: %d\n", max_level); } ; if_stmt: IF '(' condition ')' { level++; if(level > max_level) max_level = level; } stmt_block { level--; } | IF '(' condition ')' stmt_block ELSE { level++; if(level > max_level) max_level = level; } stmt_block { level--; } | other_stmt ; stmt_block: '{' stmt_list '}' | stmt ; stmt_list: stmt_list stmt | stmt ; condition: ID | NUM ; other_stmt: ID ';' | /* empty */ ; %% int yyerror(char *s) { printf("Error: %s\n", s); return 0; } int main() { printf("Enter nested IF statements:\n"); yyparse(); return 0; }

LEX File (work.l)

%{ #include "y.tab.h" %} %% "if" { return IF; } "else" { return ELSE; } [a-zA-Z][a-zA-Z0-9]* { return ID; } [0-9]+ { return NUM; } [\(\)\{\};] { return yytext[0]; } [ \t\n] { ; } . { ; } %% int yywrap() { return 1; } /* Compilation & Execution: yacc -d work.y lex work.l gcc lex.yy.c y.tab.c ./a.out */

Practical 13: Evaluate Arithmetic Expression

Aim: Evaluate an arithmetic expression involving operators +, -, *, /.

YACC File (prac13.y)

%{ #include int result; %} %token NUM %left '+' '-' %left '*' '/' %% expr: expr '+' expr { $$ = $1 + $3; } | expr '-' expr { $$ = $1 - $3; } | expr '*' expr { $$ = $1 * $3; } | expr '/' expr { $$ = $1 / $3; } | '(' expr ')' { $$ = $2; } | NUM { $$ = $1; } ; %% int yyerror(char *s) { printf("Error: %s\n", s); return 0; } int main() { printf("Enter an arithmetic expression:\n"); yyparse(); printf("Result: %d\n", result); return 0; }

LEX File (prac13.l)

%{ #include "y.tab.h" extern int yylval; %} %% [0-9]+ { yylval = atoi(yytext); return NUM; } [\+\-\*\/\(\)] { return yytext[0]; } [ \t\n] { ; } . { ; } %% int yywrap() { return 1; } /* Compilation & Execution: yacc -d prac13.y lex prac13.l gcc lex.yy.c y.tab.c ./a.out */

Important LEX & YACC Commands

LEX Compilation

lex filename.l          # Generate lex.yy.c
gcc lex.yy.c            # Compile
./a.out                 # Execute

YACC Compilation

yacc -d filename.y      # Generate y.tab.c and y.tab.h
lex filename.l          # Generate lex.yy.c
gcc lex.yy.c y.tab.c    # Compile both files
./a.out                 # Execute

LEX Regular Expressions

[a-z]       Match any lowercase letter
[A-Z]       Match any uppercase letter
[0-9]       Match any digit
[a-zA-Z]    Match any letter
.           Match any character except newline
*           Match 0 or more occurrences
+           Match 1 or more occurrences
?           Match 0 or 1 occurrence
^           Beginning of line
$           End of line

YACC Grammar Notation

%token      Define tokens
%left       Left associative operators
%right      Right associative operators
%%          Grammar rules section
$$          Semantic value of LHS
$1, $2      Semantic values of RHS symbols
